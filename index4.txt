const state = {
  status: 'starting',
  lastMessage: '-',
  connectedAt: null,
  logs: []
}

function stringifyMsg(msg) {
  try {
    if (!msg) return 'Unknown message'
    if (typeof msg === 'string') return msg

    if (msg.text) return msg.text

    if (Array.isArray(msg.extra)) {
      return msg.extra.map(x => x.text || '').join('')
    }

    if (typeof msg.toString === 'function') {
      const str = msg.toString()
      if (str !== '[object Object]') return str
    }

    return JSON.stringify(msg)
  } catch {
    return 'Unknown message'
  }
}

function escapeHtml(str) {
  return String(str)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
}

function getUptime() {
  if (!state.connectedAt) return '-'

  const sec = Math.floor((Date.now() - state.connectedAt) / 1000)
  const h = Math.floor(sec / 3600)
  const m = Math.floor((sec % 3600) / 60)
  const s = sec % 60

  return `${h}h ${m}m ${s}s`
}

function setMessage(msg) {
  state.lastMessage = msg

  const line = `[${new Date().toLocaleTimeString()}] ${msg}`
  state.logs.unshift(line)

  if (state.logs.length > 40) {
    state.logs = state.logs.slice(0, 40)
  }

  console.log(line)
}

function scheduleReconnect() {
  if (reconnectTimeout) return

  reconnectTimeout = setTimeout(() => {
    reconnectTimeout = null
    createBot()
  }, 5000)
}

function createBot() {
  try {
    if (bot) {
      bot.removeAllListeners()
      bot.quit()
    }
  } catch {}

  state.status = 'connecting'
  state.connectedAt = null
  setMessage('กำลังเชื่อมต่อ...')

  bot = mineflayer.createBot({
    host: config.host,
    port: config.port,
    username: config.username,
    version: false,
    physicsEnabled: false
  })

  bot.once('spawn', () => {
    state.status = 'online'
    state.connectedAt = Date.now()

    setMessage('เข้าเซิร์ฟแล้ว กำลังล็อกอิน...')

    setTimeout(() => {
      if (!bot) return

      try {
        bot.chat('/login a12345')
        setMessage('ส่งคำสั่งล็อกอินแล้ว')
      } catch (e) {
        setMessage(`ล็อกอินไม่สำเร็จ: ${e.message}`)
      }
    }, 1500)
  })

  bot.on('chat', (username, message) => {
    if (username === bot.username) return
    setMessage(`[CHAT] ${username}: ${message}`)
  })

  bot.on('whisper', (username, message) => {
    if (username === bot.username) return
    setMessage(`[WHISPER] ${username}: ${message}`)
  })

  bot.on('message', (jsonMsg) => {
    setMessage(`[SERVER] ${stringifyMsg(jsonMsg)}`)
  })

  bot.on('messagestr', (msg) => {
    setMessage(`[RAW] ${msg}`)
  })

  bot.on('kicked', (reason) => {
    state.status = 'kicked'
    setMessage(`[KICKED] ${stringifyMsg(reason)}`)
  })

  bot.on('end', () => {
    state.status = 'offline'
    setMessage('หลุดจากเซิร์ฟ กำลัง reconnect ใน 5 วินาที...')
    scheduleReconnect()
  })

  bot.on('error', (err) => {
    state.status = 'error'
    setMessage(`[ERROR] ${err?.message || String(err)}`)
  })
}

app.post('/command', (req, res) => {
  const msg = String(req.body.cmd || '').trim()

  if (msg && bot && bot.chat) {
    try {
      bot.chat(msg)
      setMessage(`ส่งข้อความ: ${msg}`)
    } catch (e) {
      setMessage(`ส่งไม่สำเร็จ: ${e.message}`)
    }
  } else {
    setMessage('บอทยังไม่พร้อมรับคำสั่ง')
  }

  res.redirect('/')
})

app.post('/reconnect', (req, res) => {
  try {
    bot.quit()
  } catch {}

  setMessage('สั่ง reconnect จาก dashboard')
  createBot()

  res.redirect('/')
})
